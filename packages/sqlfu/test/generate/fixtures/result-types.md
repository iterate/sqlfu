## emits named param types and a nullable single-row result for limit 1 queries

<details>
<summary>input</summary>

```sql (definitions.sql)
create table posts (id integer primary key, slug text not null, title text);
```

```sql (sql/find-post-by-slug.sql)
select id, slug, title from posts where slug = :slug limit 1;
```

</details>

<details>
<summary>output</summary>

```ts (sql/.generated/find-post-by-slug.sql.ts)
import type {Client} from 'sqlfu';

const sql = `select id, slug, title from posts where slug = ? limit 1;`;
const query = (params: findPostBySlug.Params) => ({
	name: "findPostBySlug",
	sql,
	args: [params.slug],
});

export const findPostBySlug = Object.assign(
	async function findPostBySlug(client: Client, params: findPostBySlug.Params): Promise<findPostBySlug.Result | null> {
		const rows = await client.all<findPostBySlug.Result>(query(params));
		return rows.length > 0 ? rows[0] : null;
	},
	{ sql, query },
);

export namespace findPostBySlug {
	export type Params = {
		slug: string;
	};
	export type Result = {
		id: number;
		slug: string;
		title?: string;
	};
}
```

```sql (sql/find-published-post-by-slug.sql)

select id, published_at from posts where published_at is not null limit 1;
```

```sql (sql/list-sessions.sql)
select s.id, u.id from sessions s join users u on u.id = s.user_id;
```

## makes columns from a left-joined subquery or parenthesised join nullable

<details data-outputs="sql/.generated/queries.sql.ts">
<summary>input</summary>

```sql (definitions.sql)
create table posts (post_id integer primary key, title text not null);
create table post_stats (post_id integer not null, views integer not null);
create table post_flags (post_id integer not null, flagged integer not null);
```

```sql (sql/queries.sql)
/** @name listPostsWithStats */
select p.title, s.views
from posts p
left join (select post_id, views from post_stats) s on s.post_id = p.post_id;

/** @name listPostsWithStatsAndFlags */
select *
from posts
left join (post_stats join post_flags using (post_id)) using (post_id);
```

</details>
